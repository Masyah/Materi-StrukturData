Hashing¶

Hashing adalah teknik yang digunakan dalam struktur data yang secara efisien menyimpan dan mengambil data dengan cara yang memungkinkan akses cepat.

Hashing melibatkan pemetaan data ke indeks tertentu dalam tabel hash (array item) menggunakan fungsi hash . Ini memungkinkan pengambilan informasi yang cepat berdasarkan kuncinya.

Keunggulan hashing adalah kita dapat melakukan ketiga operasi (pencarian, penyisipan, dan penghapusan) dalam waktu O(1) rata-rata.

Hashing terutama digunakan untuk mengimplementasikan sekumpulan item yang berbeda (hanya kunci) dan kamus (pasangan kunci-nilai).

Berikut contoh hashing menggunakan metode modulo. Fungsi hash H(x) = x % 10mengubah bilangan besar apa pun menjadi nilai yang lebih kecil antara 0 dan 9.

def remainderFunction(data, num):
    return (data % num)

def createHashTable(num):
    temp = []
    for i in range(num):
        temp.append([]) # Menggunakan list kosong untuk Chaining
    return temp

def putData(data, table):
    for i in range(len(data)):
        ind = remainderFunction(data[i], len(table))
        table[ind].append(data[i]) # Menambahkan data ke dalam list (chain)
    return table

def updateHash(dataLama, dataBaru, table):
    hashVal = remainderFunction(dataLama, len(table))
    keranjang = table[hashVal]
    found = False
    
    for i in range(len(keranjang)):
        if keranjang[i] == dataLama:
            keranjang[i] = dataBaru
            found = True
            print("Data", dataLama, "berhasil diubah menjadi", dataBaru)
            break
            
    if not found:
        print("Data", dataLama, "tidak ditemukan untuk diupdate.")
    return table

def deleteHash(dataHapus, table):
    hashVal = remainderFunction(dataHapus, len(table))
    keranjang = table[hashVal]
    found = False
    
    for i in range(len(keranjang)):
        if keranjang[i] == dataHapus:
            del keranjang[i]
            found = True
            print("Data", dataHapus, "berhasil dihapus.")
            break
            
    if not found:
        print("Data", dataHapus, "tidak ditemukan untuk didelete.")
    return table

tabelKu = createHashTable(5)

dataInput = [10, 15, 20, 25, 11]

print("Tabel awal:", tabelKu) 

updateHash(20, 99, tabelKu)
print("Tabel setelah update:", tabelKu)

deleteHash(15, tabelKu)
print("Tabel setelah delete:", tabelKu)

def searchHash(data, table):
    hashVal = remainderFunction(data, len(table))
    for item in table[hashVal]:
        if item == data:
            return True
    return False

print("Apakah 99 ada?", searchHash(99, tabelKu))

# Persiapan Tabel (Sama seperti kodemu yang awal)
def createHashTable(num):
    temp = []
    for i in range(num):
        temp.append('none')
    return temp

def fungsi_hash(nilai, ukuran):
    return nilai % ukuran

# --- 1. CREATE (Insert) ---
def insert_linear(nilai, tabel):
    indeks_awal = fungsi_hash(nilai, len(tabel))
    i = 0
    
    while i < len(tabel):
        # Geser indeks (Linear: + i)
        indeks_cek = (indeks_awal + i) % len(tabel)
        
        # Bisa diisi kalau slotnya 'none' atau 'deleted'
        if tabel[indeks_cek] == 'none' or tabel[indeks_cek] == 'deleted':
            tabel[indeks_cek] = nilai
            print("Berhasil insert", nilai, "di indeks", indeks_cek)
            return tabel
            
        i += 1 # Lanjut geser ke sebelah
        
    print("Tabel penuh! Gagal insert", nilai)
    return tabel

# --- 2. READ (Search) ---
def search_linear(nilai, tabel):
    indeks_awal = fungsi_hash(nilai, len(tabel))
    i = 0
    
    while i < len(tabel):
        indeks_cek = (indeks_awal + i) % len(tabel)
        
        if tabel[indeks_cek] == nilai:
            return True, indeks_cek # Ketemu
            
        # Kalau ketemu 'none', berarti datanya memang tidak ada (berhenti mencari)
        if tabel[indeks_cek] == 'none':
            break
            
        # Kalau ketemu 'deleted', program tetap lanjut geser karena mungkin datanya ada di depan
        i += 1
        
    return False, -1 # Tidak ketemu

# --- 3. UPDATE (Cari lalu Ganti) ---
def update_linear(nilai_lama, nilai_baru, tabel):
    # Manfaatkan fungsi search di atas untuk mencari lokasinya
    ketemu, indeks = search_linear(nilai_lama, tabel)
    
    if ketemu == True:
        tabel[indeks] = nilai_baru # Langsung timpa nilainya
        print("Update berhasil:", nilai_lama, "menjadi", nilai_baru)
    else:
        print("Update gagal:", nilai_lama, "tidak ditemukan.")
    return tabel

# --- 4. DELETE (Cari lalu tandai 'deleted') ---
def delete_linear(nilai_dihapus, tabel):
    ketemu, indeks = search_linear(nilai_dihapus, tabel)
    
    if ketemu == True:
        tabel[indeks] = 'deleted' # Jangan diganti 'none', tapi 'deleted'
        print("Delete berhasil:", nilai_dihapus, "sudah dihapus.")
    else:
        print("Delete gagal:", nilai_dihapus, "tidak ditemukan.")
    return tabel

def midSquareHash(nilai, ukuran_tabel):
    kuadrat = nilai * nilai
    
    str_kuadrat = str(kuadrat)
    panjang = len(str_kuadrat)
    
    tengah = panjang // 2 
    
    if panjang % 2 != 0:
        nilai_tengah = str_kuadrat[tengah]
    else:
        nilai_tengah = str_kuadrat[tengah - 1 : tengah + 1]
        
    indeks_akhir = int(nilai_tengah) % ukuran_tabel
    
    print("Proses:", nilai, "-> Dikuadratkan:", kuadrat, "-> Diambil tengah:", nilai_tengah, "-> Indeks akhir:", indeks_akhir)
    return indeks_akhir
